# defines the package follows the #.
#

# Temurin 26.0.1 gives the same list as 21. The list is every unqualified export, under the prefixes
# above, of each module in the boot layer whose class loader is not the boot loader:
#
#   java --add-modules ALL-SYSTEM NonBootPackages.java
#
#   for (Module m : ModuleLayer.boot().modules())
#     if (m.getClassLoader() != null)
#       for (ModuleDescriptor.Exports e : m.getDescriptor().exports())
#         if (!e.isQualified()) print(e.source().replace('.', '/') + "/  # " + m.getName());
#
# Regenerate it when the build JDK changes. On JDK 8 these packages are either in rt.jar or not on
# the bootstrap class path at all; the JDK 9+ rule is the stricter one.
